  function normalize(name) {
    return String(name || "").replace(/^GP_/, "").replace(/_/g, "").toLowerCase();

  function resolve(name) {
    var want = normalize(name);
    var target = bridge || gpRef;
    if (target && typeof target[name] === "function")
      return target[name].bind(target);
    var obj = target;
    while (obj && obj !== Object.prototype) {
      var names = Object.getOwnPropertyNames(obj);
      for (var i = 0; i < names.length; i++) {
        var key = names[i];
        if (typeof obj[key] !== "function") continue;
        if (normalize(key) === want) return obj[key].bind(target);
      }
      obj = Object.getPrototypeOf(obj);
    }
    if (gpRef && want === "gameready" && typeof gpRef.gameStart === "function")
      return gpRef.gameStart.bind(gpRef);
    return null;
  }

  function invoke(method, args) {
    var fn = resolve(method);
    if (!fn) {
      send({ type: "log", arg: "missing JS method " + method });
      return;
    }
    try {
      args = args || [];
      var mapped = [];
      for (var i = 0; i < args.length; i++) {
        var a = args[i];
        if (a === "True") mapped.push(true);
        else if (a === "False") mapped.push(false);
        else if (typeof a === "string" && /^-?\d+$/.test(a)) mapped.push(Number(a));
        else mapped.push(a);
      }
      var result = fn.apply(bridge || gpRef, mapped);
      if (result && typeof result.then === "function") {
        result.catch(function (err) {
          send({ type: "log", arg: String(err && err.message || err) });
        });
      } else if (result !== undefined) {
        send({ type: "snapshot", entries: [{ k: method, v: toUnity(result) }] });
      }
    } catch (err) {
      send({ type: "log", arg: String(err && err.message || err) });
    }
  }

  function toUnity(value) {
    switch (typeof value) {
      case "boolean":
      case "number":
        return String(value);
      case "object":
        try {
          if (value instanceof Map) return JSON.stringify(Array.from(value.values()));
          return JSON.stringify(value);
        } catch (e) { return ""; }
      case "undefined":
        return "";
      default:
        return String(value);
    }
  }

      }

        done(!!(res && res.ok));
      }).catch(function () { done(false); });
